<!-- Form to Send Data -->
<form id="data-form">
  <input type="text" id="item-title" placeholder="Enter item title..." required />
  <button type="submit">Add Item</button>
</form>

<!-- Container to Display Fetched Data -->
<ul id="item-list"></ul>

<script type="module">
  import { fetchData, sendData } from './supabaseClient.js'

  const dataForm = document.getElementById('data-form')
  const itemTitleInput = document.getElementById('item-title')
  const itemList = document.getElementById('item-list')

  // 1. Fetch and render data when page loads
  document.addEventListener('DOMContentLoaded', loadAndRenderData)

  async function loadAndRenderData() {
    itemList.innerHTML = 'Loading...'
    const items = await fetchData()

    itemList.innerHTML = '' // Clear list
    items.forEach(item => {
      const li = document.createElement('li')
      li.textContent = item.title // Replace 'title' with your column name
      itemList.appendChild(li)
    })
  }

  // 2. Send new data when form is submitted
  dataForm.addEventListener('submit', async (e) => {
    e.preventDefault()

    const newRecord = {
      title: itemTitleInput.value // Match column key with table schema
    }

    const result = await sendData(newRecord)
    if (result) {
      itemTitleInput.value = '' // Clear input
      loadAndRenderData()       // Refresh displayed list
    }
  })
</script>
